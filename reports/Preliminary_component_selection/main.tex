\section{Design Requirements Summary}

\begin{itemize}
\item Base dimensions: $800 \times 600 \times 420$ mm 
\item Ground clearance: 40 mm 
\item Drive architecture: differential drive 
\item Drive wheels: 2 centrally positioned wheels 
\item Initial wheel diameter: 180 mm 
\item Passive support: 2 caster wheels 
\item Initial caster diameter: 100 mm 
\item Initial module interface area: $500 \times 400$ mm 
\item Module interface position: centred on the upper platform 
\item Electrical baseline: 24 V 
\item Normal speed: 0.8 to 1.2 m/s 
\item Maximum design speed: 1.5 m/s
\end{itemize}

\section{Power System}

\subsection{Battery Pack}

\textbf{Role:} Provides the primary onboard power for the complete AMR-X system, supporting differential-drive motors, compute, sensors, and safety systems for $\geq 3$ hours continuous operation under nominal load ($1.0$ m/s cruise, 50 kg payload).

\textbf{Specifications to Meet:}
\begin{itemize}
\item Nominal Voltage: $24$ V DC
\item Capacity: $20$ Ah ($480$ Wh) minimum for Maxon EC45 motors (upgraded from $15$ Ah)
\item Chemistry: LiFePO$_4$ preferred for safety and cycle life
\item Built-in BMS: Over-current, over-temperature, over/under-voltage protection
\item Discharge Connectors: Suitable for continuous $30$--$50$ A draw and peak $80$--$100$ A transients
\end{itemize}

\subsection{Power Distribution and DC-DC Converters}

\textbf{Role:} Routes $24$ V battery power safely to all subsystems (motors, compute, sensors, safety) with appropriate overcurrent protection, voltage conversion, and electromagnetic interference filtering.

\textbf{Specifications to Meet:}
\begin{itemize}
\item Main Fuse: $80$ A (battery short-circuit protection)
\item Motor Power Rail: $30$ A fused (motor driver protection)
\item Compute Regulator: $24$ V $\rightarrow$ $12$ V at $5$ A (Jetson input)
\item Sensor Regulator: $24$ V $\rightarrow$ $5$ V at $5$ A (all sensors and safety beacon)
\item Wiring: 4/6/8/10 AWG depending on current rating
\item EMI Filtering: Ferrite clamps on motor lines
\end{itemize}

\begin{longtable}{|p{4cm}|p{10cm}|}
\hline
\textbf{Parameter} & \textbf{Details} \\
\hline

Component &
$24$ V Power Distribution System with Protection and DC-DC Conversion \\
\hline

Function &
Safely distributes battery power to three independent rails: motor power ($24$ V direct, $30$ A), compute power ($24$ V $\rightarrow$ $12$ V regulated), and sensor power ($24$ V $\rightarrow$ $5$ V regulated). Includes overcurrent protection, EMI filtering, and voltage regulation. Supports higher current draw from Maxon EC45 motors (vs. JGA25-370). \\

\item Connectors: XT90 (battery), Anderson PowerPole (distribution)
